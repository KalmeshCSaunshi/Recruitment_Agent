def get_exact_token_count(text):
    encoding = tiktoken.get_encoding("cl100k_base") 
    num_tokens = len(encoding.encode(text)) 
    return num_tokens

def extract_text_from_resume(filepath):
    text = ""
    try:
        if filepath.lower().endswith('.docx') or filepath.lower().endswith('.doc'):
            text = docx2txt.process(filepath)
        elif filepath.lower().endswith('.pdf'):
            with fitz.open(filepath) as doc:
                for page in doc:
                    page_text = page.get_text()
                    if page_text:
                        text += page_text + "\n"
            
            if len(text.strip()) < 50:
                images = convert_from_path(filepath)
                for image in images:
                    page_text = pytesseract.image_to_string(image)
                    text += page_text + "\n"
    except Exception as e:
        print(f"Error extracting document: {e}")
    return text


def calculate_skill_experience(resume_text, skill_keyword):

    text = resume_text.replace("–", "-").replace("—", "-")

    current_date = datetime.now()

    total_months = 0

    # Split by experience/project sections
    experience_blocks = re.split(
        r'\n(?=[A-Z][A-Za-z0-9 &(),.-]+\s+[A-Za-z]{3,9}\s\d{4}\s*-\s*(?:Present|[A-Za-z]{3,9}\s\d{4}))',
        text
    )

    counted_ranges = set()

    for block in experience_blocks:

        if skill_keyword.lower() not in block.lower():
            continue

        date_match = re.search(
            r'([A-Za-z]{3,9}\s\d{4})\s*-\s*(Present|[A-Za-z]{3,9}\s\d{4})',
            block
        )

        if not date_match:
            continue

        start_str = date_match.group(1)
        end_str = date_match.group(2)

        try:

            start_date = None
            end_date = None

            for fmt in ("%b %Y", "%B %Y"):

                try:
                    start_date = datetime.strptime(start_str, fmt)
                    break
                except:
                    pass

            if end_str.lower() == "present":
                end_date = current_date
            else:
                for fmt in ("%b %Y", "%B %Y"):

                    try:
                        end_date = datetime.strptime(end_str, fmt)
                        break
                    except:
                        pass

            if not start_date or not end_date:
                continue

            # Prevent duplicate counting
            range_key = (start_date, end_date)

            if range_key in counted_ranges:
                continue

            counted_ranges.add(range_key)

            months = (
                (end_date.year - start_date.year) * 12
                + (end_date.month - start_date.month)
            )

            total_months += max(months, 0)

        except:
            continue

    years = round(total_months / 12, 1)

    return str(years)

@app.route('/api/source', methods=['POST'])
def source():
    if 'jd' not in request.form: return jsonify({"error": "Missing JD"}), 400
    jd = request.form['jd']
    page_num = int(request.form.get('page', 1))
    keywords = sourcing_agent.get_search_keywords(jd)
    try:
        raw_leads = asyncio.run(sourcing_agent.perform_authenticated_search(
            keywords['boolean_query'], 
            min_exp=keywords.get('min_exp', 2), 
            max_exp=keywords.get('max_exp', 8),
            page_num=page_num
        ))
    except: raw_leads = []
    
    def analyze_lead(lead):
        prompt = f"""
            Analyze this profile against the JD.

            JD:
            {jd}

            Candidate:
            {lead['name']}

            Context:
            {lead['context']}

            Return STRICT JSON:
            {{
                "match_percentage": number,
                "matched_skills": ["skill1", "skill2"],
                "context_evidence": "short explanation"
            }}
        """
        for attempt in range(3):
            try:
                payload = {"model": MODEL_NAME, "prompt": prompt, "stream": False, "format": "json"}
                response = requests.post(OLLAMA_API_URL, json=payload, timeout=90)
                response.raise_for_status()
                analysis = json.loads(response.json().get('response'))
                lead.update(analysis)
                return lead
            except Exception as e:
                print(f"Ollama analyze_lead attempt {attempt + 1} failed: {e}")
                if attempt == 2:
                    lead.update({
                        "match_percentage": 50,
                        "matched_skills": [],
                        "context_evidence": "Analysis failed"
                    })
                import time
                time.sleep(1)
        return lead

    with concurrent.futures.ThreadPoolExecutor(max_workers=5) as executor:
        analyzed_leads = list(executor.map(analyze_lead, raw_leads))
    
    return jsonify({
        "keywords_used": keywords.get("primary_keywords", []),
        "boolean_query": keywords.get("boolean_query", ""),
        "search_url": "https://resdex.naukri.com/v3",
        "simulated_leads": analyzed_leads
    })
